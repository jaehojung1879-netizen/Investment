def yahoo(sym):
    o=json.loads(gzip.decompress((R/f'data/raw/yahoo_{sym}.json.gz').read_bytes()))['chart']['result'][0]
    tz=o['meta']['exchangeTimezoneName']
    idx=pd.to_datetime(o['timestamp'],unit='s',utc=True).tz_convert(tz).normalize().tz_localize(None)
    q=o['indicators']['quote'][0]
    f=pd.DataFrame({'close':q['close'],'adj':o['indicators']['adjclose'][0]['adjclose']},index=idx)
    f['div']=0.0
    for d in o.get('events',{}).get('dividends',{}).values():
        t=pd.Timestamp(d['date'],unit='s',tz='UTC').tz_convert(tz).normalize().tz_localize(None)
        f.loc[t,'div']+=d['amount']
    f=f.sort_index().dropna(subset=['close'])
    f['tr']=((f.close+f['div'])/f.close.shift()).fillna(1).cumprod()
    return f,o

def fred(sym):
    f=pd.read_csv(R/f'data/raw/fred_{sym}.csv.gz',index_col=0,parse_dates=True)
    return pd.to_numeric(f.iloc[:,0],errors='coerce').dropna()

assets={}

checks['crossVendor']={}

    checks['crossVendor'][sym]={'sessions':len(diff),'first':str(diff.index[0].date()),'last':str(diff.index[-1].date()),

        summary[name]={'calculated':{k:float(v) for k,v in calculated.items()},'errorsVsSaved':diffs,
